🚗 1. Line Following Robot

Goal: Build a robot that automatically follows a black line using IR sensors.

Components Required

  • Arduino UNO
  • 2 IR Sensors
  • L298N Motor Driver
  • 2 DC Motors
  • Robot Chassis
  • Battery Pack
  • Jumper Wires

Working Principle

The IR sensors detect the black line. The Arduino continuously reads both sensors and controls the motors to keep the robot on the track.

Applications

  • Factory Automation
  • Warehouse Robots
  • Hospital Delivery Robots

Arduino Code


// Line Following Robot

int leftSensor=2;
int rightSensor=3;

int motorA1=4;
int motorA2=5;
int motorB1=6;
int motorB2=7;

void setup(){

pinMode(leftSensor,INPUT);
pinMode(rightSensor,INPUT);

pinMode(motorA1,OUTPUT);
pinMode(motorA2,OUTPUT);
pinMode(motorB1,OUTPUT);
pinMode(motorB2,OUTPUT);

}

void loop(){

int L=digitalRead(leftSensor);
int R=digitalRead(rightSensor);

if(L==LOW && R==LOW){

forward();

}

else if(L==LOW){

left();

}

else if(R==LOW){

right();

}

else{

stopRobot();

}

}

void forward(){

digitalWrite(motorA1,HIGH);
digitalWrite(motorA2,LOW);
digitalWrite(motorB1,HIGH);
digitalWrite(motorB2,LOW);

}

void left(){

digitalWrite(motorA1,LOW);
digitalWrite(motorA2,HIGH);
digitalWrite(motorB1,HIGH);
digitalWrite(motorB2,LOW);

}

void right(){

digitalWrite(motorA1,HIGH);
digitalWrite(motorA2,LOW);
digitalWrite(motorB1,LOW);
digitalWrite(motorB2,HIGH);

}

void stopRobot(){

digitalWrite(motorA1,LOW);
digitalWrite(motorA2,LOW);
digitalWrite(motorB1,LOW);
digitalWrite(motorB2,LOW);

}

🚧 2. Obstacle Avoiding Robot

Goal: Detect obstacles using an ultrasonic sensor and automatically change direction.

Components Required

  • Arduino UNO
  • HC-SR04 Ultrasonic Sensor
  • L298N Motor Driver
  • 2 DC Motors
  • Robot Chassis
  • Battery

Working Principle

The ultrasonic sensor measures the distance to objects. If an obstacle is detected within a set distance, the robot stops, turns, and continues moving.

Applications

  • Autonomous Vehicles
  • Delivery Robots
  • Cleaning Robots

Arduino Code


// Obstacle Avoiding Robot

// Ultrasonic Sensor Pins
const int trigPin = 9;
const int echoPin = 10;

// Motor Driver Pins
const int motorA1 = 4;
const int motorA2 = 5;
const int motorB1 = 6;
const int motorB2 = 7;

long duration;
int distance;

void setup() {

  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  pinMode(motorA1, OUTPUT);
  pinMode(motorA2, OUTPUT);
  pinMode(motorB1, OUTPUT);
  pinMode(motorB2, OUTPUT);

}

void loop() {

  distance = getDistance();

  if(distance > 20){

    forward();

  }

  else{

    stopRobot();
    delay(300);

    turnRight();
    delay(600);

  }

}

// Measure Distance

int getDistance(){

  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(trigPin, LOW);

  duration = pulseIn(echoPin, HIGH);

  distance = duration * 0.034 / 2;

  return distance;

}

// Move Forward

void forward(){

  digitalWrite(motorA1, HIGH);
  digitalWrite(motorA2, LOW);

  digitalWrite(motorB1, HIGH);
  digitalWrite(motorB2, LOW);

}

// Turn Right

void turnRight(){

  digitalWrite(motorA1, HIGH);
  digitalWrite(motorA2, LOW);

  digitalWrite(motorB1, LOW);
  digitalWrite(motorB2, HIGH);

}

// Stop Robot

void stopRobot(){

  digitalWrite(motorA1, LOW);
  digitalWrite(motorA2, LOW);

  digitalWrite(motorB1, LOW);
  digitalWrite(motorB2, LOW);

Components Required

  • Arduino UNO
  • HC-SR04 Sensor
  • Servo Motor
  • Dustbin
  • Battery

Working Principle

The ultrasonic sensor detects a nearby hand. The Arduino rotates the servo motor to open the lid and closes it automatically after a few seconds.

Applications

  • Smart Homes
  • Hospitals
  • Schools

Arduino Code


// Smart Dustbin Using Arduino

#include 

Servo lid;

// Ultrasonic Sensor Pins
const int trigPin = 9;
const int echoPin = 10;

long duration;
int distance;

void setup() {

  lid.attach(6);      // Servo connected to pin 6
  lid.write(0);       // Lid closed

  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

}

void loop() {

  distance = getDistance();

  if (distance > 0 && distance <= 20) {

    lid.write(90);      // Open lid
    delay(3000);        // Keep open for 3 seconds
    lid.write(0);       // Close lid
    delay(1000);

  }

  delay(100);

// Function to measure distance

  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

💡 4. Automatic Street Light

Goal: Switch street lights ON at night and OFF during the day using an LDR sensor.

Components Required

  • Arduino UNO
  • LDR Sensor
  • 220Ω Resistor
  • LED
  • Breadboard

Working Principle

The LDR senses ambient light. When it becomes dark, the Arduino turns the LED ON. During daylight, it turns the LED OFF.

Arduino Code


// Automatic Street Light Using LDR

const int ldrPin = A0;      // LDR connected to Analog Pin A0
const int ledPin = 8;       // LED connected to Digital Pin 8

int ldrValue = 0;

void setup() {

  pinMode(ledPin, OUTPUT);
  Serial.begin(9600);

}

void loop() {

  // Read LDR value
  ldrValue = analogRead(ldrPin);

  Serial.print("LDR Value: ");
  Serial.println(ldrValue);

  // If it is dark, turn ON the LED
  if (ldrValue < 500) {

    digitalWrite(ledPin, HIGH);

  }

  // If it is bright, turn OFF the LED
  else {

    digitalWrite(ledPin, LOW);

  }

  delay(200);

}

🌱 5. Automatic Plant Watering System

Goal: Water plants automatically when the soil becomes dry.

Components Required

  • Arduino UNO
  • Soil Moisture Sensor
  • Relay Module
  • Mini Water Pump
  • Battery

Working Principle

The soil moisture sensor checks the moisture level. When the soil is dry, the Arduino turns on the water pump until the moisture reaches the desired level.

Applications

  • Smart Agriculture
  • Gardens
  • Greenhouses

Arduino Code


// Automatic Plant Watering System

const int soilSensor = A0;     // Soil Moisture Sensor
const int relayPin = 8;        // Relay Module

int moistureValue;

void setup() {

  pinMode(relayPin, OUTPUT);
  digitalWrite(relayPin, HIGH);   // Pump OFF (active LOW relay)

  Serial.begin(9600);

}

void loop() {

  moistureValue = analogRead(soilSensor);

  Serial.print("Soil Moisture: ");
  Serial.println(moistureValue);

  // If soil is dry
  if (moistureValue > 600) {

    Serial.println("Soil Dry - Pump ON");

    digitalWrite(relayPin, LOW);   // Turn Pump ON
    delay(5000);                   // Water for 5 seconds

    digitalWrite(relayPin, HIGH);  // Turn Pump OFF

  }

  else {

    Serial.println("Soil Wet - Pump OFF");

    digitalWrite(relayPin, HIGH);

  }

  delay(2000);

}

🔐 4. RFID Door Lock System

Goal: Unlock a door using an authorized RFID card.

  • RC522 RFID Module
  • RFID Cards
  • Servo Motor
  • Buzzer
  • LEDs

Working Principle

The RFID reader scans the card. If the card is authorized, Arduino unlocks the door using a servo motor.

Applications

  • Office Security
  • Attendance Systems
  • Smart Locks

Arduino Code


// RFID Door Lock System
// Arduino UNO + RC522 + Servo Motor

#include 
#include 
#include 

#define SS_PIN 10
#define RST_PIN 9

MFRC522 rfid(SS_PIN, RST_PIN);
Servo door;

const int buzzer = 7;

// Change this UID to your RFID Card UID
byte allowedUID[4] = {0xDE, 0xAD, 0xBE, 0xEF};

void setup() {

  SPI.begin();
  rfid.PCD_Init();

  door.attach(6);
  door.write(0);          // Door Locked

  pinMode(buzzer, OUTPUT);

  Serial.begin(9600);
  Serial.println("Scan RFID Card");

}

void loop() {

  if (!rfid.PICC_IsNewCardPresent())
    return;

  if (!rfid.PICC_ReadCardSerial())
    return;

  bool access = true;

  for (byte i = 0; i < 4; i++) {

    if (rfid.uid.uidByte[i] != allowedUID[i]) {

      access = false;
      break;

    }

  }

  if (access) {

    Serial.println("Access Granted");

    digitalWrite(buzzer, HIGH);
    delay(200);
    digitalWrite(buzzer, LOW);

    door.write(90);      // Unlock Door
    delay(5000);

    door.write(0);       // Lock Door

  }

  else {

    Serial.println("Access Denied");

    for (int i = 0; i < 3; i++) {

      digitalWrite(buzzer, HIGH);
      delay(150);
      digitalWrite(buzzer, LOW);
      delay(150);

    }

  }

  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();

}

Goal: Detect empty parking spaces and automatically control the parking gate.

Components Required

  • Arduino UNO
  • IR Sensors
  • Servo Motor
  • LCD Display
  • LEDs
  • Buzzer

Working Principle

IR sensors detect vehicle presence. Arduino updates the available parking spaces on the LCD and opens or closes the gate automatically.

Applications

  • Shopping Malls
  • Airports
  • Hospitals
  • Smart Cities

Arduino Code


// Smart Parking System
// Arduino UNO + IR Sensors + Servo + LCD

#include 
#include 

Servo gate;

// LCD Pins: RS, EN, D4, D5, D6, D7
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

// IR Sensor Pins
const int entrySensor = 6;
const int exitSensor  = 7;

// Servo Pin
const int servoPin = 8;

int availableSlots = 5;

void setup() {

  pinMode(entrySensor, INPUT);
  pinMode(exitSensor, INPUT);

  gate.attach(servoPin);
  gate.write(0);          // Gate Closed

  lcd.begin(16, 2);

  lcd.setCursor(0, 0);
  lcd.print("SMART PARKING");

  delay(2000);
  lcd.clear();

}

void loop() {

  lcd.setCursor(0, 0);
  lcd.print("Slots Left:");

  lcd.setCursor(12, 0);
  lcd.print(availableSlots);

  // Vehicle Entering
  if (digitalRead(entrySensor) == LOW && availableSlots > 0) {

    gate.write(90);      // Open Gate
    delay(3000);

    gate.write(0);       // Close Gate

    availableSlots--;

  }

  // Vehicle Leaving
  if (digitalRead(exitSensor) == LOW && availableSlots < 5) {

    gate.write(90);

    delay(3000);

    gate.write(0);

    availableSlots++;

  }

  delay(200);

}
